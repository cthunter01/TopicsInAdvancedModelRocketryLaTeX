\chapter{Correspondence Between Metric and English Units}\label{appA}

\section{Centimeter-Gram-Second (CGS) to English Units}\label{appA:sec:1}
\begin{align*}
  \text{1 gram (g.)} &= .0353\ \text{ounce-mass}\\
                     &= .0022\ \text{pound-mass}\\
                     &= 6.84\times 10^{-5}\ \text{slug}\\
  \text{1 dyne (dn.)} &= 2.248\times 10^{-6}\ \text{pound-force}\\
  \text{1 centimeter (cm.)} &= .0328\ \text{foot}\\
                            &= .3937\ \text{inch}
\end{align*}

\section{Meter-Kilogram-Second (MKS) to English Units}\label{appA:sec:2}
\begin{align*}
  \text{1 kilogram (Kg.)} &= 35.27\ \text{ounce-mass}\\
                          &= 2.204\ \text{pound-mass}\\
                          &= .0684\ \text{slug}\\
  \text{1 newton (nt.)} &= .2248\ \text{pound-force}\\
  \text{1 meter (m.)} &= 3.28\ \text{feet}\\
                      &= 39.37\ \text{inches}
\end{align*}
