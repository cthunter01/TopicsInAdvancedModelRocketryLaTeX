% The trajectory family (Figs 10-14): open axes, equal x and y scales as printed (the axes 4.2 in wide), the printed
% ticks; each axis line runs half a tick step past its last label where a curve does (here the apex of (a), 831 m);
% curves (a), (b), (c) solid in s1, s2, s3, each tag 2.75 mm out along the normal of its descending branch
% (its centre computed by fig13.py, in fig13-marks.csv); times \footnotesize in ink on leaders.

% \timemark{<x>}{<y>}{<angle>}{<length>}{<label anchor>}{<time>}: a leader from the point on the curve to the time
\newcommand{\timemark}[6]{%
  \draw[leader] (axis cs:#1,#2) -- ++(#3:#4) node[anchor=#5, font=\footnotesize, inner sep=1.2pt] {#6};}
\begin{document}
\begin{tikzpicture}
  \begin{axis}[tamr,
      x=0.0035in, y=0.0035in,
      xmin=0, xmax=1200, ymin=0, ymax=900,
      xtick distance=200, ytick={0,200,...,800},
      xlabel={$x$ (m)}, ylabel={$y$ (m)}]
    \addplot[series1, solid] table[x=x, y=y, col sep=comma] {figures/v2/ch4/fig13-a.csv};
    \addplot[series2, solid] table[x=x, y=y, col sep=comma] {figures/v2/ch4/fig13-b.csv};
    \addplot[series3, solid] table[x=x, y=y, col sep=comma] {figures/v2/ch4/fig13-c.csv};

    \timemark{\Aax}{\Aay}{-100}{4.5mm}{north}{9.40}
    \timemark{\Aix}{0}{40}{5.5mm}{south west}{26.47}
    \timemark{\Bax}{\Bay}{-100}{4.5mm}{north}{9.00}

    \timemark{\Cax}{\Cay}{30}{6.5mm}{south west}{3.80}
    \timemark{\Cix}{0}{25}{6mm}{south west}{9.40}
    \node[curve tag] at (axis cs:\Atx,\Aty) {a};
    \node[curve tag] at (axis cs:\Btx,\Bty) {b};
    \node[curve tag] at (axis cs:\Ctx,\Cty) {c};
  \end{axis}
