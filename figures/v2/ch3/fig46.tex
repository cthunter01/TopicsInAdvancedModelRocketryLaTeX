% Chapter 3, Figure 46: S_s/S_m against fineness ratio ell/d_m for the cylinder and three nose shapes (scan:
% figures/ch3/fig46.png). Curves: fig46.py -> fig46-cylinder.csv, -ellipsoid.csv, -ogive.csv, -cone.csv,
% each with two columns:
%   exact   (the default, \exacttrue): cylinder eq. (170) 4 ell/d_m; ellipsoid eq. (171a); ogive eq. (171c)
%           2.67 ell/d_m (the book gives no exact ogive formula); cone eq. (171d);
%   printed (\exactfalse): the straight lines the 1973 figure draws, which the caption calls approximate:
%           4 ell/d_m, pi ell/d_m (the asymptote of (171a); eq. (171b) prints 1 + pi ell/d_m), 2.67 ell/d_m
%           (171c) and 2 ell/d_m (171e).
% The noses start at ell/d_m = 1.5, as printed (the caption: terminated at their lower limit). Measured on
% the scan, the drawn lines are 4.00, 3.15, 2.68 and 2.03 ell/d_m (fig46.calib.json).
\documentclass[11pt]{standalone}
\usepackage{tamrfig}
\newif\ifexact \exacttrue
\begin{document}
\begin{tikzpicture}
  \ifexact \def\col{exact}\else \def\col{printed}\fi
  \begin{axis}[tamr grid, grid=both,
      width=4.2in, height=2.8in,
      xmin=0, xmax=6, xtick={0,...,6}, minor xtick={0.5,1.5,...,5.5},
      ymin=0, ymax=16, ytick={0,2,...,16}, minor ytick={1,3,...,15},
      xlabel={$\dfrac{\ell}{d_m}$},
      ylabel={$\dfrac{S_s}{S_m}$}, ylabel style={rotate=-90, anchor=east}]
    \pgfplotsinvokeforeach{cylinder, ellipsoid, ogive, cone}{
      \addplot[series1] table[x=f, y=\col, col sep=comma] {figures/v2/ch3/fig46-#1.csv};
    }
    % direct labels, along each line on its upper side, as printed (white knock-outs over the grid only)
    \begin{scope}[every node/.style={fill=white, inner sep=1pt, sloped, above=1.5pt}]
      \path (axis cs:2.45,9.8) -- (axis cs:3.05,12.2) node[midway] {Cylinder};
      \path (axis cs:3.2,10.11) -- (axis cs:3.8,12.0) node[midway] {Ellipsoid};
      \path (axis cs:4.05,10.81) -- (axis cs:4.55,12.15) node[midway] {Ogive};
      \path (axis cs:5.0,10.0) -- (axis cs:5.5,11.0) node[midway] {Cone};
    \end{scope}
  \end{axis}

\end{document}
