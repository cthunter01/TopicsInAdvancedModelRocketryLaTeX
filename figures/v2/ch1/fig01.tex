% Chapter 1, Figure 1: translation (a) and rotation (b) (scan: figures/ch1/fig01.png, PDF 40).
% The model rocket of Figures 6-8, nose up, C.G. 65 percent of the length back from the nose.
\documentclass[11pt]{standalone}
\usepackage{tamrfig}
\begin{document}
\begin{tikzpicture}
  \def\L{4.4}\def\cgf{0.65}
  % \uprocket{<C.G. position>}{<tilt, degrees clockwise>}: the rocket, nose up, with its C.G. mark
  % (a pic takes its position, not its orientation, from an enclosing rotation: both are given here)
  \newcommand{\uprocket}[3][]{%
    \pic[rotate=-90-#3] at ($#2+({90-#3}:{\cgf*\L})$) {rocket={model, length=\L, diameter=0.3,
        nose length=0.85, root chord=0.7, tip chord=0.3, span=0.45, sweep=0.4, #1}};}
  % (a) translation: from position 1 to position 2, without rotation
  \coordinate (cg1) at (0,0); \coordinate (cg2) at (5.2,2.5);
  \uprocket{(cg1)}{0} \uprocket{(cg2)}{0}
  \draw[thin vec, shorten <=3pt, shorten >=3.5pt] (cg1) -- (cg2);
  \node[cg mark] at (cg1) {}; \node[cg mark] at (cg2) {};
  \node[font=\small] at ($(cg1)+(0.45,1.3)$) {1};
  \node[font=\small] at ($(cg2)+(0.45,1.3)$) {2};
  \node[panel] at (5.9,-2.7) {(a)};
  % (b) rotation about the C.G., without translation
  \begin{scope}[shift={(9.3,0.4)}]
    \coordinate (cg) at (0,0);
    \draw[thin line] (0,{-(1-\cgf)*\L-0.7}) -- (0,{\cgf*\L+1.6});
    \draw[centerline] (-120:{(1-\cgf)*\L+0.6}) -- (60:{\cgf*\L+1.6});
    \uprocket[centerline=false]{(cg)}{0} \uprocket[centerline=false]{(cg)}{30}
    \draw[thin vec] (90:{\cgf*\L+1.2}) arc[start angle=90, end angle=60, radius={\cgf*\L+1.2}];
    \node[cg mark] at (cg) {};
    \node[font=\small] at (-0.45,1.3) {1};
    \node[font=\small] at (1.55,1.4) {2};
    \node[panel] at (1.9,-3.1) {(b)};
  \end{scope}
\end{tikzpicture}
\end{document}
