% Chapter 2, Figure 21: underdamped response to an impulse (scan: figures/ch2/fig21.png).
% Curve: fig21.py, eq. (38) with D and omega by eqs. (16), (17), zeta = 0.25. Units (no scale is printed):
% t in 1/omega_n, alpha_X in H/(I_L omega_n), the frame shared with Figs 22 and 23 (one rocket, three
% dampings; the initial slope H/I_L, eq. (37b), is 1; y axis to -0.42 in all three, formulas beneath it). Marks
% t_m, alpha_Xm by eqs. (41a), (41b), joined to the peak by solid lines as printed (peak line); the line
% "Slope = H/I_L" is the tangent at the origin. The formulas are eqs. (41b), (41a) as lettered.
\documentclass[11pt]{standalone}
\usepackage{tamrfig}
\begin{document}
\begin{tikzpicture}
  \pgfmathsetmacro\z{0.25}
  \pgfmathsetmacro\w{sqrt(1-\z^2)}                       % omega / omega_n, eq. (17)
  \pgfmathsetmacro\tm{rad(atan(\w/\z))/\w}                % eq. (41a)
  \pgfmathsetmacro\am{exp(-\z*\tm)*sin(deg(\w*\tm))/\w}   % eq. (41b)
  \pgfmathsetmacro\ts{0.86}                              % top of the tangent line
  \begin{axis}[tamr sketch, name=ax,
      x=0.46in, y=1.8in, xmin=0, xmax=8.6, ymin=-0.42, ymax=1.0,
      xlabel={$t$ (sec)}, ylabel={$\alpha_X$ (rad)},
      xtick={\tm}, xticklabels={$t_m$}, ytick={0,\am}, yticklabels={$0$, $\alpha_{Xm}$},
      hide obscured y ticks=false,
      every axis y label/.style={at={(axis cs:0,1.0)}, anchor=south, inner sep=2pt}]
    \draw[peak line] (axis cs:0,\am) -- (axis cs:\tm,\am) -- (axis cs:\tm,0);
    \addplot[series2, solid, domain=0:\ts, samples=2] {x};
    \addplot[series1] table[x=t, y=alpha, col sep=comma] {figures/v2/ch2/fig21.csv};
    \node[anchor=west, inner sep=2pt] at (axis cs:\ts,\ts) {Slope $= \dfrac{H}{I_L}$};
  \end{axis}
  \node[anchor=north west, font=\small] at ([xshift=0.35in, yshift=-6pt]ax.south west)
    {$\begin{aligned}
      \alpha_{Xm} &= \frac{H}{I_L\omega}\,e^{-\frac{D}{\omega}\arctan\left(\frac{\omega}{D}\right)}
        \sin\left[\arctan\left(\frac{\omega}{D}\right)\right]\\[4pt]
      t_m &= \frac{\arctan\left(\frac{\omega}{D}\right)}{\omega}
    \end{aligned}$};
\end{tikzpicture}
\end{document}
