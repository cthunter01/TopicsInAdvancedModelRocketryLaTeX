% Chapter 2, Figure 25: amplitude ratio against frequency ratio (scan: figures/ch2/fig25.png).
% Curves: eq. (48b), AR = 1/(C_1 sqrt((beta^2 - 1)^2 + (2 zeta beta)^2)), plotted in units of 1/C_1.
\documentclass[11pt]{standalone}
\usepackage{tamrfig}
\begin{document}
\begin{tikzpicture}
  \begin{axis}[tamr,
      width=4.2in, height=3.0in,
      xmin=0, xmax=2, xtick={0,0.25,...,2},
      xticklabels={0, 0.25, 0.50, 0.75, 1.00, 1.25, 1.50, 1.75, 2.00},
      ymin=0, ymax=3, ytick={0,1,2,3}, minor ytick={0.5,1.5,2.5},
      yticklabels={0, $\dfrac{1}{C_1}$, $\dfrac{2}{C_1}$, $\dfrac{3}{C_1}$},
      xlabel={$\beta$},
      ylabel={$\mathit{AR}$ [rad/(dyn-cm)]},
      every node near coord/.append style={font=\footnotesize}]
    % reference levels: AR = 1/C_1 (its value at beta = 0) and beta = 1
    \draw[guide] (axis cs:0,1) -- (axis cs:2,1);
    \draw[guide] (axis cs:1,0) -- (axis cs:1,3);
    % zeta = 0: 1/|beta^2 - 1|, leaving the frame at 3/C_1
    \addplot[series1, domain=0:0.8165, samples=80] {1/(1-x^2)};
    \addplot[series1, domain=1.1547:2, samples=80] {1/(x^2-1)};
    \pgfplotsinvokeforeach{0.2, 0.5, 0.70710678, 1, 2}{
      \addplot[series1, domain=0:2, samples=161] {1/sqrt((x^2-1)^2 + (2*#1*x)^2)};
    }
    % direct labels: on the curve (a white knock-out) where no other curve passes, else beside it
    \begin{scope}[font=\footnotesize, inner sep=1pt]
      \node[anchor=west] at (axis cs:1.19,2.75) {$\zeta = 0$};
      \node[anchor=south, fill=white] at (axis cs:0.965,2.58) {$\zeta = .2$};
      \node[anchor=south] at (axis cs:0.66,1.17) {$\zeta = .5$};
      \node[fill=white] at (axis cs:1.06,0.667) {$\zeta = \sqrt{2}/2$};
      \node[fill=white] at (axis cs:0.62,0.722) {$\zeta = 1$};
      \node[fill=white] at (axis cs:0.30,0.664) {$\zeta = 2$};
    \end{scope}
  \end{axis}
\end{tikzpicture}
\end{document}
