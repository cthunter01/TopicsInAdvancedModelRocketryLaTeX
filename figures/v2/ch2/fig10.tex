% Chapter 2, Figure 10: zero damping, simple harmonic motion in yaw (scan: figures/ch2/fig10.png).
% Curve: fig10.py, eq. (15) with D = 0 (C_2 = 0): alpha_X = A sin(omega_n t + phi), A and phi by eqs. (18), (19);
% alpha_X0 = 0.8, Omega_X0 = 2.12 in the family's units (time 1/omega_n): phi = 0.361 rad, A = 2.83 alpha_X0 (the
% drawn tick ratio). The instants a-f (fig10-marks.csv) are the maxima, zeros and minimum,
% omega_n t + phi = pi/2 ... 3pi; each rocket below stands under its instant, tilted about its C.G. (not to the
% scale of the graph: 13 degrees for A). The straight line is the tangent at t = 0, slope Omega_X0.
% Layout shared by Figs 10-14 (tamr sketch; 0.4 in per 1/omega_n, 0.576 in per unit of yaw angle; this frame
% runs from -3.1 to hold the dimensions under the trough, as printed).
\documentclass[11pt]{standalone}
\usepackage{tamrfig}
\pgfplotsset{free response/.style={tamr sketch, x=0.4in, y=0.576in,
    xmin=0, xmax=9.6, ymin=-3.1, ymax=2.8,
    xlabel={$t$ (sec)}, ylabel={$\alpha_X$ (rad)},
    % the label above the arrow, upright (tamr sketch's rotate=-90 turns it on its side under axis lines=middle)
    every axis y label/.style={at={(ticklabel* cs:1.0)}, anchor=south, inner sep=3pt}}}
\tikzset{tangent/.style={s2, line width=1pt}}
\pgfplotstableread[col sep=comma]{figures/v2/ch2/fig10-marks.csv}\marks
\foreach \key in {alpha0,Omega0,A,ta,tb,tc,td,te,tf} {%
  \pgfplotstablegetelem{0}{\key}\of\marks \expandafter\xdef\csname m@\key\endcsname{\pgfplotsretval}}
\makeatletter
\newcommand{\mk}[1]{\csname m@#1\endcsname}
\makeatother
% the rocket of the lower row (cm): length, radius, nose, fin root (= sweep: delta fins), span; C.G. at 0.685 L
\def\L{3.43}\def\rad{0.122}\def\nl{0.67}\def\cr{0.67}\def\sp{0.33}
\pgfmathsetmacro\cgn{0.685*\L}   % nose tip to C.G.
\pgfmathsetmacro\cgt{\L-\cgn}    % C.G. to tail
\def\R{2.75}                     % radius of the angle arc about the C.G. (above the nose)
\def\tilt{13}                    % the drawn yaw A (degrees)
% \yawrocket{<coordinate of the C.G.>}{<tilt, degrees, + = nose right>}{<angle label or empty>}
\newcommand{\yawrocket}[3]{%
  \begin{scope}[shift={(#1)}]
    \pic[rotate={-90-(#2)}] at ({\cgn*sin(#2)},{\cgn*cos(#2)})
      {rocket={length=\L, diameter={2*\rad}, nose length=\nl, root chord=\cr, tip chord=0, span=\sp,
               sweep=\cr, centerline=false}};
    \draw[centerline] ({(\R+0.15)*sin(#2)},{(\R+0.15)*cos(#2)}) -- ({-(\cgt+0.55)*sin(#2)},{-(\cgt+0.55)*cos(#2)});
    \ifdim #2pt=0pt\else
      \draw[extension] (0,{\R+0.15}) -- (0,{-(\cgt+0.55)});
      \draw[angle arc] (0,0) ++(90:\R) arc[start angle=90, end angle={90-(#2)}, radius=\R];
      \node[anchor=south, font=\small, inner sep=2pt] at ({90-(#2)/2}:{\R+0.03}) {#3};
    \fi
    \node[cg mark] at (0,0) {};
  \end{scope}}
% \instant{<letter>}{<alpha>}{<placement of the tag>}: the point on the curve, its tag, and a coordinate at the
% foot of the axis for the rocket below
\newcommand{\instant}[3]{%
  \node[point] (p#1) at (axis cs:\mk{t#1},#2) {};
  \node[curve tag, #3] at (p#1) {#1};
  \coordinate (x#1) at (axis cs:\mk{t#1},-3.1);}
\begin{document}
\begin{tikzpicture}
  \begin{axis}[free response, ytick={\mk{alpha0},\mk{A}}, yticklabels={$\alpha_{X0}$, $A$}]
    \draw[guide] (axis cs:0,\mk{A}) -- (axis cs:9.35,\mk{A});
    \addplot[series1] table[x=t, y=alpha, col sep=comma] {figures/v2/ch2/fig10.csv};
    % tangent at t = 0: alpha_X0 + Omega_X0 t
    \pgfmathsetmacro\tt{0.85}
    \draw[tangent] (axis cs:0,\mk{alpha0}) -- (axis cs:\tt,{\mk{alpha0}+\mk{Omega0}*\tt})
      node[anchor=west, text=ink, font=\small, inner sep=2pt] {Slope${}=\Omega_{X0}$};
    \node[anchor=east, font=\footnotesize, text=ink2, inner sep=3pt] at (axis cs:0,0) {0};
    % the instants a-f on the curve, lettered as the rockets below
    % (a macro, not a loop: pgfplots runs the drawing commands at the end of the axis)
    % ((f) inside the last lobe, left of f, as printed: clear of the t-axis arrowhead and of the curve)
    \instant{a}{\mk{A}}{below=4pt}  \instant{b}{0}{above right=4pt}  \instant{c}{-\mk{A}}{above=4pt}
    \instant{d}{0}{above left=4pt}  \instant{e}{\mk{A}}{below=4pt}  \instant{f}{0}{xshift=-12pt, yshift=8pt}
    % time to the first zero and the period
    \draw[extension] (axis cs:\mk{tb},0) -- (axis cs:\mk{tb},-2.9);
    \draw[extension] (axis cs:\mk{tf},0) -- (axis cs:\mk{tf},-2.9);
    \dimline{(axis cs:0,-2.75)}{(axis cs:\mk{tb},-2.75)}{$\dfrac{\pi-\varphi}{\omega_n}$}
    \dimline{(axis cs:\mk{tb},-2.75)}{(axis cs:\mk{tf},-2.75)}{$\dfrac{2\pi}{\omega_n}$}
  \end{axis}
  % the rocket at each instant, viewed from the negative y axis: tilted by A, 0, -A, 0, A, 0
  \foreach \c/\th/\lab in {a/\tilt/$A$, b/0/, c/-\tilt/$-A$, d/0/, e/\tilt/$A$, f/0/} {
    \coordinate (cg\c) at ($(x\c)+(0,{-0.75-\R})$);
    \yawrocket{cg\c}{\th}{\lab}
    \node[curve tag] at ($(cg\c)+(0,{-\cgt-0.95})$) {\c};
  }
\end{tikzpicture}
\end{document}
