% Chapter 2, Figure 43: measurement of moments of inertia with a torsion wire (scan: figures/ch2/fig43.png).
% (a) the rocket hung horizontally from the wire (longitudinal moment of inertia); (b) hung nose down from the
% tail (radial moment of inertia). The circled lowercase panel letters of the art are kept, at the lower right of
% each panel on one baseline, in the house curve tag (standing rule 2: the text cites "43A", "43B").
% A true orthographic view, elevation 32 deg, azimuth 45 deg (the house 3D view of Ch2 Fig 3): TikZ x along
% the beam (towards its free end, drawn down to the right), y across the beam, z up. Lengths are the scan's
% pixels measured along the page horizontal (a two-by-four 64 wide and 32 deep; the rocket 177 long, 9.9 in
% diameter), drawn 1.1 times the printed size. Hidden lines by painting back to front with white fills.
% Each panel: the overhanging beam (cut by the panel's edge), a C-clamp with a knurled screw holding the upper
% "T" (crossbar along the beam, overhanging its end), the wire, and the rocket with its lower "T": in (a) under
% two tape bands at 0.68 of the length; in (b) across the tail, taped to two fins.
\documentclass[11pt]{standalone}
\usepackage{tamrfig}
% projection: 1 unit = 1 scan pixel (1/150 in) x 1.1; x -> (a,-b), y -> (a,b), z -> (0,h)
\pgfmathsetmacro\ua{1.1*2.54/150}
\pgfmathsetmacro\ub{\ua*sin(32)}
\pgfmathsetmacro\uh{\ua*cos(32)/cos(45)}
% \figblk{x0}{x1}{y0}{y1}{z0}{z1}: a rectangular block, its three visible faces (+x, -y, +z)
\newcommand{\figblk}[6]{%
  \filldraw[outline, fill=white] (#2,#3,#5) -- (#2,#4,#5) -- (#2,#4,#6) -- (#2,#3,#6) -- cycle
    (#1,#3,#5) -- (#2,#3,#5) -- (#2,#3,#6) -- (#1,#3,#6) -- cycle
    (#1,#3,#6) -- (#2,#3,#6) -- (#2,#4,#6) -- (#1,#4,#6) -- cycle;}
% \figcylz{x}{y}{z0}{z1}{r}: an upright cylinder, top visible (silhouettes at 45 and 225 deg)
\newcommand{\figcylz}[5]{%
  \filldraw[outline, fill=white] plot[domain=225:405, samples=37, variable=\t] ({#1+#5*cos(\t)}, {#2+#5*sin(\t)}, #3)
    -- plot[domain=45:225, samples=37, variable=\t] ({#1+#5*cos(\t)}, {#2+#5*sin(\t)}, #4) -- cycle;
  \draw[outline] plot[domain=0:360, samples=61, variable=\t] ({#1+#5*cos(\t)}, {#2+#5*sin(\t)}, #4) -- cycle;}
% the rocket, in its own frame (s from the nose tip to the tail, p, q radial), set by the enclosing scope:
% \rkt{phi_s}{back fin angles}{front fin angles}; phi_s = silhouette angle, phi_s + 90 faces the viewer
\def\rL{177}\def\rR{4.95}\def\rNL{33}\def\rCR{42}\def\rCT{13}\def\rSP{18}\def\rSW{29}
\pgfmathsetmacro\rrho{(\rR*\rR+\rNL*\rNL)/(2*\rR)}
\newcommand{\rfin}[1]{%
  \filldraw[outline, fill=white] ({\rL-\rCR}, {\rR*cos(#1)}, {\rR*sin(#1)})
    -- ({\rL-\rCR+\rSW}, {(\rR+\rSP)*cos(#1)}, {(\rR+\rSP)*sin(#1)})
    -- ({\rL-\rCR+\rSW+\rCT}, {(\rR+\rSP)*cos(#1)}, {(\rR+\rSP)*sin(#1)})
    -- (\rL, {\rR*cos(#1)}, {\rR*sin(#1)}) -- cycle;}
\newcommand{\rarc}[4]{% s, radius, from, to
  plot[domain=#3:#4, samples=37, variable=\t] (#1, {(#2)*cos(\t)}, {(#2)*sin(\t)})}
\newcommand{\rkt}[3]{%
  \pgfmathsetmacro\pa{#1}\pgfmathsetmacro\pb{#1+180}\pgfmathsetmacro\pc{#1-180}\pgfmathsetmacro\pd{#1+360}%
  \foreach \f in {#2} {\rfin{\f}}
  \filldraw[outline, fill=white]
    plot[domain=0:\rNL, samples=40, variable=\s] (\s, {(sqrt(max(0,\rrho*\rrho-(\rNL-\s)^2))+\rR-\rrho)*cos(\pa)},
      {(sqrt(max(0,\rrho*\rrho-(\rNL-\s)^2))+\rR-\rrho)*sin(\pa)})
    -- plot[domain=\pa:\pc, samples=37, variable=\t] (\rL, {\rR*cos(\t)}, {\rR*sin(\t)})
    -- plot[domain=\rNL:0, samples=40, variable=\s] (\s, {(sqrt(max(0,\rrho*\rrho-(\rNL-\s)^2))+\rR-\rrho)*cos(\pb)},
      {(sqrt(max(0,\rrho*\rrho-(\rNL-\s)^2))+\rR-\rrho)*sin(\pb)}) -- cycle;
  \draw[outline] \rarc{\rNL}{\rR}{\pa}{\pb};
  \draw[outline] \rarc{\rL}{\rR}{0}{360} -- cycle;
  \draw[thin line] \rarc{\rL}{0.74*\rR}{0}{360} -- cycle;
  \draw[thin line] \rarc{\rL}{0.28*\rR}{0}{360} -- cycle;
  \foreach \f in {#3} {\rfin{\f}}}
% a tape band round the body at s (width 3): the halves of its edges that face the viewer
\newcommand{\rband}[1]{%
  \filldraw[thin line, fill=white] \rarc{#1-1.5}{\rR+0.45}{\pa}{\pb}
    -- plot[domain=\pb:\pa, samples=37, variable=\t] (#1+1.5, {(\rR+0.45)*cos(\t)}, {(\rR+0.45)*sin(\t)}) -- cycle;}
% the beam, C-clamp, knurled screw and upper "T"; the beam's end face at x = 0, the wire at (7, 31, 32)
% \apparatus{<clip: left, top in screen units>}
\newcommand{\apparatus}[2]{%
  % C-clamp frame in the plane x = -25 ... -13: spine y 82-92, arms z -8..0 and 40..48 from y = 22
  \def\cprof{(-13,22,48) -- (-13,92,48) -- (-13,92,-8) -- (-13,22,-8) -- (-13,22,0) -- (-13,82,0) -- (-13,82,40)
    -- (-13,22,40) -- cycle}
  \filldraw[outline, fill=white] \cprof;
  \filldraw[outline, fill=white] (-25,82,0) -- (-13,82,0) -- (-13,82,40) -- (-25,82,40) -- cycle;
  \begin{scope}
    \clip[x=\ua cm, y=\ua cm] (#1,-4) rectangle (70,#2);
    \figblk{-600}{0}{0}{64}{0}{32}
  \end{scope}
  % the part of the clamp above the beam, painted again over it
  \fill[white] (-13,22,48) -- (-13,92,48) -- (-13,92,32) -- (-13,82,32) -- (-13,82,40) -- (-13,22,40) -- cycle;
  \draw[outline] (-13,92,32) -- (-13,92,48) -- (-13,22,48) -- (-13,22,40) -- (-13,82,40) -- (-13,82,32);
  \filldraw[outline, fill=white] (-25,82,32) -- (-13,82,32) -- (-13,82,40) -- (-25,82,40) -- cycle;
  \filldraw[outline, fill=white] (-25,22,40) -- (-13,22,40) -- (-13,22,48) -- (-25,22,48) -- cycle;
  \filldraw[outline, fill=white] (-25,22,48) -- (-13,22,48) -- (-13,92,48) -- (-25,92,48) -- cycle;
  % upper "T": crossbar along the beam, overhanging the end, under the swivel pad
  \draw[line width=1.1pt, line cap=butt] (-30,31,32.7) -- (44,31,32.7);
  \figblk{-26}{-12}{24}{38}{33.4}{35.8}
  \draw[outline] (-12,31,35.8) -- (-12,38,35.8);
  % screw: threaded rod above the arm, knurled head
  \figcylz{-19}{31}{48}{71}{3.5}
  \foreach \zz in {50.5,53,...,69} \draw[hair] plot[domain=225:405, samples=19, variable=\t]
    ({-19+3.5*cos(\t)}, {31+3.5*sin(\t)}, {\zz+0.9*sin(\t-45)});
  \figcylz{-19}{31}{71}{82}{6.7}
  \foreach \t in {235,250,...,395} \draw[hair] ({-19+6.7*cos(\t)}, {31+6.7*sin(\t)}, 71.8) -- ++(0,0,9.4);
}
